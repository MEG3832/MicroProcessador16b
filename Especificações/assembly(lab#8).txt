Validação - Crivo de Eratóstenes
Tabela inicia no endereço 111 | Fim de loop via AND (bit 5)
=====================================================================

# Inicialização de Constantes

0. LD R4, 1             # Passo unitário
1. LD R5, 32            # Limite superior do loop

# Preencher RAM com os números de 1 a 32 (endereços 112 a 143)

2. LD R2, 1             # Índice do loop
3. LD R1, 112           # Endereço

4. MOV A1, R2           # Teste de parada com AND (bit 5 setado)
5. AND A1, R5        
6. BEQ 1                # Se (N AND 32) == 0 pula o JUMP pro próximo loop
7. JUMP 17

8. MOV A1, R2           # A1 = Dado
9. SW  A1, R1           # RAM[R1] <= Dado

10. MOV A1, R1
11. ADD A1, R4          # Endereço++
12. MOV R1, A1
13. MOV A1, R2
14. ADD A1, R4
15. MOV R2, A1          # Índice++
16. JUMP 4              # Volta pro início do loop

# Eliminar múltiplos de 2 (zerar a partir de N = 4)

17. LD R2, 4            # Índice
18. LD R1, 115          # 111 + 4 = 115
19. LD R3, 2            # Passo = 2

20. MOV A1, R2
21. AND A1, R5          # A1 = N AND 32
22. BEQ 1               # Se (N AND 32) == 0, pula O JUMP pro próximo loop
23. JUMP 33

24. LD  A1, 0
25. SW  A1, R1          # RAM[R1] <= 0

26. MOV A1, R2
27. ADD A1, R3          # Índice += 2
28. MOV R2, A1
29. MOV A1, R1
30. ADD A1, R3          # Endereço += 2
31. MOV R1, A1        
32. JUMP 20             # Volta pro início do loop

# Eliminar múltiplos de 3 (zerar a partir de N = 6)

33. LD R2, 6            # Índice
34. LD R1, 117          # 111 + 6 = 117
35. LD R3, 3            # Passo = 3

36. MOV A1, R2
37. AND A1, R5          # A1 = N AND 32
38. BEQ 1               # Se (N AND 32) == 0, pula o JUMP pro próximo loop
39. JUMP 49

40. LD  A1, 0
41. SW  A1, R1          # RAM[R1] <= 0

42. MOV A1, R2
43. ADD A1, R3          # Índice += 3
44. MOV R2, A1
45. MOV A1, R1
46. ADD A1, R3          # Endereço += 3
47. MOV R1, A1
48. JUMP 36             # Volta pro início do loop

# Eliminar múltiplos de 5 (zerar a partir de N = 10)

49. LD R2, 10           # Índice
50. LD R1, 121          # 111 + 10 = 121
51. LD R3, 5            # Passo = 5

52. MOV A1, R2
53. AND A1, R5          # A1 = N AND 32
54. BEQ 1               # Se (N AND 32) == 0, pula o JUMP pro próximo loop
55. JUMP 65

56. LD  A1, 0     
57. SW  A1, R1          # RAM[R1] <= 0

58. MOV A1, R2
59. ADD A1, R3          # Índice += 5
60. MOV R2, A1
61. MOV A1, R1
62. ADD A1, R3          # Endereço += 5
63. MOV R1, A1
64. JUMP 52             # Volta pro início do loop

# Leitura dos primos

65. LD R2, 2            # Índice
66. LD R1, 113          # Endereço

67. MOV A1, R2
68. AND A1, R5          # A1 = N AND 32
69. BEQ 1
70. JUMP 79             # Encerra o programa após ler N = 32

71. LW A1, R1           # A1 <= RAM[R1]

72. MOV A1, R1
73. ADD A1, R4
74. MOV R1, A1          # Endereço++
75. MOV A1, R2
76. ADD A1, R4
77. MOV R2, A1          # Índice++
