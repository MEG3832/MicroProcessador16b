=====================================================================
Validação - Crivo de Eratóstenes
Tabela inicia no endereço 111 | Fim de loop via AND (bit 5)
=====================================================================

# Inicialização de Constantes

1. LD R4, 1             # Passo unitário
2. LD R5, 32            # Limite superior do loop

# Preencher RAM com os números de 1 a 32 (endereços 112 a 143)

3. LD R2, 1             # Índice do loop
4. LD R1, 112           # Endereço

5. MOV A1, R2           # Teste de parada com AND (bit 5 setado)
6. AND A1, R5        
7. BEQ 3                # Se (N AND 32) == 0 pula o JUMP pro próximo loop
8. JUMP 18

9. MOV A1, R2           # A1 = Dado
10. SW  A1, R1          # RAM[R1] <= Dado

11. MOV A1, R1
12. ADD A1, R4          # Endereço++
13. MOV R1, A1
14. MOV A1, R2
15. ADD A1, R4
16. MOV R2, A1          # Índice++
17. JUMP 5              # Volta pro início do loop

# Eliminar Múltiplos de 2 (zerar a partir de N = 4)

18. LD R2, 4            # Índice
19. LD R1, 115          # 111 + 4 = 115
20. LD R3, 2            # Passo = 2

21. MOV A1, R2
22. AND A1, R5          # A1 = N AND 32
23. BEQ 3               # Se (N AND 32) == 0, pula O JUMP pro próximo loop
24. JUMP 34

25. LD  A1, 0
26. SW  A1, R1          # RAM[R1] <= 0

27. MOV A1, R2
28. ADD A1, R3          # Índice += 2
29. MOV R2, A1
30. MOV A1, R1
31. ADD A1, R3          # Endereço += 2
32. MOV R1, A1        
33. JUMP 21             # Volta pro início do loop

# Eliminar Múltiplos de 3 (zerar a partir de N = 6)

34. LD R2, 6            # Índice
35. LD R1, 117          # 111 + 6 = 117
36. LD R3, 3            # Passo = 3

37. MOV A1, R2
38. AND A1, R5          # A1 = N AND 32
39. BEQ 3               # Se (N AND 32) == 0, pula o JUMP pro próximo loop
40. JUMP 50

41. LD  A1, 0
42. SW  A1, R1          # RAM[R1] <= 0

43. MOV A1, R2
44. ADD A1, R3          # Índice += 3
45. MOV R2, A1
46. MOV A1, R1
47. ADD A1, R3          # Endereço += 3
48. MOV R1, A1
49. JUMP 37             # Volta pro início do loop

# Eliminar Múltiplos de 5 (zerar a partir de N = 10)

50. LD R2, 10           # Índice
51. LD R1, 121          # 111 + 10 = 121
52. LD R3, 5            # Passo = 5

53. MOV A1, R2
54. AND A1, R5          # A1 = N AND 32
55. BEQ 3               # Se (N AND 32) == 0, pula o JUMP pro próximo loop
56. JUMP 66

57. LD  A1, 0     
58. SW  A1, R1          # RAM[R1] <= 0

59. MOV A1, R2
60. ADD A1, R3          # Índice += 5
61. MOV R2, A1
62. MOV A1, R1
63. ADD A1, R3          # Endereço += 5
64. MOV R1, A1
65. JUMP 53             # Volta pro início do loop

# Leitura dos primos

66. LD R2, 2            # Índice
67. LD R1, 113          # Endereço

68. MOV A1, R2
69. AND A1, R5          # A1 = N AND 32
70. BEQ 3
71. JUMP 80             # Encerra o programa após ler N = 32

72. LW A1, R1          # A1 <= RAM[R1]

73. MOV A1, R1
74. ADD A1, R4
75. MOV R1, A1          # Endereço++
76. MOV A1, R2
77. ADD A1, R4
78. MOV R2, A1          # Índice++
79. JUMP 68

80. NOP                 # Fim da execução
